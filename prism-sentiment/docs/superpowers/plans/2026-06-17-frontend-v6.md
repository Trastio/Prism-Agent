### Task 2: Add dashboard-summary report endpoint

**Files:** Modify `api/routes/reports.py`

- [ ] **Step 1: Add dashboard_summary endpoint**

New endpoint at bottom of reports.py:

```python
class DashboardSummaryRequest(BaseModel):
    vehicle_id: str
    start_date: str | None = None
    end_date: str | None = None

@router.post("/dashboard-summary")
async def create_dashboard_summary(
    req: DashboardSummaryRequest,
    session: AsyncSession = Depends(get_session),
):
    vehicle = await session.get(Vehicle, req.vehicle_id)
    if not vehicle:
        raise HTTPException(404, "Vehicle not found")
    
    today = date.today()
    start_date = req.start_date or (today - timedelta(days=30)).isoformat()
    end_date = req.end_date or today.isoformat()
    start_dt = datetime.fromisoformat(start_date)
    end_dt = datetime.fromisoformat(end_date) + timedelta(days=1)
    
    # Gather dashboard data
    # 1. Overall sentiment
    sent_result = await session.execute(
        select(AnalyzedPost.sentiment, func.count(AnalyzedPost.id))
        .where(AnalyzedPost.vehicle_id == req.vehicle_id, SELF_ONLY)
        .group_by(AnalyzedPost.sentiment)
    )
    sent_counts = {r[0]: r[1] for r in sent_result.all()}
    total_analyzed = sum(sent_counts.values()) or 1
    pos_pct = round(sent_counts.get("positive", 0) / total_analyzed * 100, 1)
    neg_pct = round(sent_counts.get("negative", 0) / total_analyzed * 100, 1)
    
    # 2. Health score, total posts (from overview logic)
    total_posts = sent_counts.get("positive",0) + sent_counts.get("negative",0) + sent_counts.get("neutral",0)
    health = round(50 + (sent_counts.get("positive",0) - sent_counts.get("negative",0)) / total_posts * 50, 1)
    
    # 3. Top dimensions
    dims = await _compute_dim_sentiment(session, req.vehicle_id, start_dt, end_dt)
    best_dim = dims[0] if dims else None
    worst_dim = dims[-1] if dims else None
    
    # 4. Top events
    events_result = await session.execute(
        select(EventGroup).where(EventGroup.vehicle_id == req.vehicle_id)
        .order_by(EventGroup.post_count.desc()).limit(5)
    )
    top_events = events_result.scalars().all()
    
    # 5. Recent top posts
    posts_result = await session.execute(
        select(RawPost.title, RawPost.platform, RawPost.likes, RawPost.comments, AnalyzedPost.sentiment)
        .join(AnalyzedPost, RawPost.id == AnalyzedPost.post_id)
        .where(RawPost.vehicle_id == req.vehicle_id, SELF_ONLY)
        .order_by((RawPost.likes + RawPost.comments * 5 + RawPost.shares * 10).desc()).limit(5)
    )
    top_posts = posts_result.all()
    
    # 6. Comments per post
    total_raw = (await session.execute(select(func.count(RawPost.id)).where(RawPost.vehicle_id == req.vehicle_id))).scalar() or 1
    total_cmt = (await session.execute(select(func.count(PostComment.id)).where(PostComment.vehicle_id == req.vehicle_id))).scalar() or 0
    
    # Build prompt
    events_text = "\n".join(
        f"- {e.event_tag}: {e.post_count}帖, {e.start_date}~{e.end_date}"
        for e in top_events
    )
    posts_text = "\n".join(
        f"- 「{p[0][:50]}」{p[1]} {p[2]}赞 {p[3]}评 {p[4]}"
        for p in top_posts
    )
    dims_text = f"最强维度: {best_dim['dim']}({best_dim['score']})，最弱维度: {worst_dim['dim']}({worst_dim['score']})" if best_dim and worst_dim else ""
    
    prompt = f"""你是汽车舆情分析师。以下是{vehicle.name}在{start_date}至{end_date}的舆情数据：

## KPI
- 健康分: {health}
- 总分析帖数: {total_posts}
- 情感占比: 正面{pos_pct}% 负面{neg_pct}% 中性{round(100-pos_pct-neg_pct,1)}%
- 帖均评论: {round(total_cmt/max(total_raw,1),1)}

## 维度体检
{dims_text}

## TOP 事件
{events_text or '(暂无)'}

## 热门帖子
{posts_text or '(暂无)'}

请生成一份300字以内的舆情简报（Markdown格式），包含：总体评价、关键发现、风险提示、本周关注建议。"""

    result = await _call_llm(prompt)
    
    report = Report(
        id=str(uuid.uuid4()),
        vehicle_id=req.vehicle_id,
        type="dashboard_summary",
        title=f"{vehicle.name} 舆情简报 {end_date}",
        content=result,
        time_range_start=start_date,
        time_range_end=end_date,
    )
    session.add(report)
    await session.commit()
    
    return {"id": report.id, "content": result}
```

Add imports at top if missing: `from models.schemas import PostComment, EventGroup, SELF_ONLY`, `import uuid`.

Add a helper `_compute_dim_sentiment` that replicates the logic from the existing `/dim-sentiment` endpoint (reads AnalyzedPost.dim_sentiment, aggregates per dimension).
