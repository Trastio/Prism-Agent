```javascript
renderOpinionTagsChart() {
    const el = document.getElementById("opinion-tags-chart");
    if (!el) return;
    if (!this.opinionTags || !this.opinionTags.length) return;
    const data = [...this.opinionTags].reverse(); // smallest at top
    const max = Math.max(...data.map(d => d[1]), 1);
    const chart = echarts.init(el);
    chart.setOption({
        tooltip: {},
        grid: { top: 8, right: 40, bottom: 8, left: 80 },
        xAxis: { type: "value", show: false },
        yAxis: {
            type: "category", data: data.map(d => d[0]),
            axisLine: { show: false }, axisTick: { show: false },
            splitLine: { show: false },
            axisLabel: { color: CHART_COLORS.muted, fontSize: 11 },
        },
        series: [{
            type: "bar",
            data: data.map(d => ({
                value: d[1],
                itemStyle: {
                    color: `rgba(58,110,165,${(0.4 + 0.55 * (d[1] / max)).toFixed(2)})`,
                    borderRadius: [0, 4, 4, 0],
                },
            })),
            barWidth: "56%",
            label: { show: true, position: "right", color: CHART_COLORS.muted, fontSize: 11, fontFamily: "Inter" },
        }],
    }, true);
    chart.on("click", (params) => {
        if (params.name) openDialog("event", { event: params.name });
    });
},
```
