## 背景

现有前端只有单车"采集"按钮（`triggerCollection`，调 `/api/vehicles/{vid}/collect`），缺两个能力：
1. **全车刷新**：一键对所有 active 车跑增量采集（抓新帖 + 刷新互动数），对应 `daily_update --all`
2. **单车刷新互动数**：只更新互动数不搜新帖（便宜），对应 `refresh_vehicle_engagement`

**核心需求**：刷新后前端**所有组件都要更新**，不能后端采了新数据前端不变。

```python
# pipeline/frequency.py（新建）
import sqlite3
import os
from datetime import date, timedelta
from typing import Literal

Tier = Literal["hot", "mid", "cold"]

# P1-1：照搬 daily_update.py 真实常量（审查 v5 §9.3 逐字核验通过）
HOT_THRESHOLD = 100
MID_THRESHOLD = 30


def db_path() -> str:
    """DB 路径（P2-2：逐字一致）。"""
    return os.getenv("DB_PATH", "data/vehicle_sentiment.db")


def recent_post_counts(vehicle_ids: list[str], db_path_str: str | None = None) -> dict[str, int]:
    """同步查近30天帖数。调用方用 asyncio.to_thread 包裹。"""
    cutoff = (date.today() - timedelta(days=30)).isoformat()
    out = {vid: 0 for vid in vehicle_ids}
    try:
        conn = sqlite3.connect(db_path_str or db_path())
        try:
            rows = conn.execute(
                "SELECT vehicle_id, COUNT(*) FROM raw_posts "
                "WHERE collected_at >= ? GROUP BY vehicle_id",
                (cutoff,),
            ).fetchall()
            for vid, cnt in rows:
                if vid in out:
                    out[vid] = cnt
        finally:
            conn.close()
    except Exception as e:
        print(f"[WARN] 近30天帖数查询失败（按 cold 兜底）: {e}")
    return out


def _tier_of(cnt: int) -> Tier:
    """P1-1：照搬原逻辑，> 不是 >=。"""
    if cnt > HOT_THRESHOLD: return "hot"
    if cnt > MID_THRESHOLD: return "mid"
    return "cold"


def select_by_frequency(
    vehicles: list[dict],
    counts: dict[str, int],
    today: date,
) -> tuple[list[dict], list[tuple[str, Tier]]]:
    """频率分层：hot 日更/mid 隔日（偶数日）/cold 周一。"""
    active = [v for v in vehicles if v.get("status") == "active"]
    targets, skipped = [], []
    for v in active:
        tier = _tier_of(counts.get(v["id"], 0))
        run = (
            tier == "hot"
            or (tier == "mid" and today.day % 2 == 0)
            or (tier == "cold" and today.weekday() == 0)
        )
        if run:
            targets.append(v)
        else:
            skipped.append((v["name"], tier))
    return targets, skipped
```

# api/routes/fleet.py（新建）
from fastapi import APIRouter, Depends
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from models.database import get_session, async_session
from models.schemas import Vehicle
from pipeline.frequency import select_by_frequency, recent_post_counts, db_path
from datetime import date
import asyncio
import logging

logger = logging.getLogger(__name__)
router = APIRouter(prefix="/api/fleet", tags=["fleet"])


@router.post("/collect")
async def trigger_fleet_collect(
    session: AsyncSession = Depends(get_session),
):
    """全车增量采集（频率分层）。
    P0-A：每车独立 session。
    P0-B：管线级并发上限在 _run_full_pipeline（_PIPELINE_SEM）。
    P1-1：counts 由 to_thread 传入。
    P2-5：无 concurrency 死参数。
    """
    vehicles = (await session.execute(
        select(Vehicle).where(Vehicle.status == "active")
    )).scalars().all()
    veh_dicts = [{"id": v.id, "name": v.name, "status": v.status} for v in vehicles]
    
    counts = await asyncio.to_thread(recent_post_counts, [v["id"] for v in veh_dicts], db_path())
    targets, skipped = select_by_frequency(veh_dicts, counts, date.today())
    
    started = []
    for v in targets:
        try:
            async with async_session() as s:  # P0-A：每车独立 session
                from api.routes.collection import trigger_collection as _tc
                await _tc(v["id"], s)
            started.append(v["id"])
        except Exception as e:
            logger.warning("fleet collect %s failed: %s", v["name"], e)
    
    return {"started": started, "skipped": [{"name": n, "tier": t} for n, t in skipped]}
```

_PIPELINE_SEM = asyncio.Semaphore(3)  # P0-B：跨车共享，覆盖管线全程

from api.pipeline_state import ACTIVE_PHASES, _is_phase_stale, PipelineProgress, pipeline_progress
REFRESHING_PHASES = ACTIVE_PHASES | {"refreshing"}

async def _run_full_pipeline(vehicle_id: str, _session_factory=None):

        try:

# 新增 refresh 端点（P1-2：写 CollectionStatus 与采集同构）
@router.post("/{vehicle_id}/refresh-engagement")
async def trigger_refresh_engagement(vehicle_id: str, session: AsyncSession = Depends(get_session)):
    """单车互动刷新 + 热度重算。异步 task。
    
    P0-3：不在请求内 await。
    P1-2：写 CollectionStatus.is_running=True + lock_acquired_at（与采集同构），
          避免 /pipeline-status DB-first 短路返回 completed。
    P1-C：复用 _running_tasks 单字典。
    P1-3：判定加 _is_phase_stale 逃逸。
    P2-1：refresh 不进 _PIPELINE_SEM（轻量）。
    """
    vehicle = await session.get(Vehicle, vehicle_id)
    if not vehicle:
        raise HTTPException(404, detail="Vehicle not found")
    
    phase_info = pipeline_phases.get(vehicle_id, {})
    if phase_info.get("phase") in REFRESHING_PHASES and not _is_phase_stale(phase_info):
        return {"vehicle_id": vehicle_id, "status": "already_running", "phase": phase_info["phase"]}
    
    # P1-2：写 CollectionStatus（与采集同构），避免 DB-first 短路
    cs = (await session.execute(
        select(CollectionStatus).where(
            CollectionStatus.vehicle_id == vehicle_id,
            CollectionStatus.source == "all",
        )
    )).scalar_one_or_none()
    if cs:
        cs.is_running = True
        cs.lock_acquired_at = datetime.now()  # task GC 修复加的字段
    else:
        cs = CollectionStatus(vehicle_id=vehicle_id, source="all", mode="initial",
                              is_running=True, lock_acquired_at=datetime.now())
        session.add(cs)
    await session.commit()
    
    pipeline_phases[vehicle_id] = {"phase": "refreshing", "posts_refreshed": 0, "error": None, "_updated_at": datetime.now()}
    # P1-3：重置 pipeline_progress
    pipeline_progress[vehicle_id] = PipelineProgress()
    pipeline_progress[vehicle_id].overall_phase = "refreshing"
    pipeline_progress[vehicle_id].current_step = "刷新互动数"
    
    # P1-C：复用 _running_tasks 单字典
    old = _running_tasks.get(vehicle_id)
    if old and not old.done():
        old.cancel()
    task = asyncio.create_task(_run_refresh_pipeline(vehicle_id))
    _running_tasks[vehicle_id] = task
    task.add_done_callback(lambda t: _running_tasks.pop(vehicle_id, None))
    return {"vehicle_id": vehicle_id, "status": "started", "phase": "refreshing"}


# P1-2 + P1-3：refresh 管线 finally 复位 CollectionStatus
async def _run_refresh_pipeline(vehicle_id: str):
    """异步：互动刷新 + 热度重算。不调 LLM。
    P1-2：finally 复位 CollectionStatus.is_running=False（与 _run_full_pipeline 同款）。
    P2-1：不进 _PIPELINE_SEM（轻量）。
    """
    from scripts.refresh_engagement import refresh_vehicle_engagement
    from pipeline.analysis.heat_calculator import HeatMetricCalculator
    _sf = async_session
    try:
        async with _sf() as session:
            n = await refresh_vehicle_engagement(session, vehicle_id)
            pipeline_phases[vehicle_id] = {"phase": "calculating", "posts_refreshed": n, "_updated_at": datetime.now()}
            pipeline_progress[vehicle_id].overall_phase = "calculating"
            pipeline_progress[vehicle_id].current_step = "重算热度"
            calc = HeatMetricCalculator(session)
            end = date.today()
            start = end - timedelta(days=90)
            await calc.calculate_range(vehicle_id, start, end)
            pipeline_phases[vehicle_id] = {"phase": "completed", "posts_refreshed": n, "_updated_at": datetime.now()}
            pipeline_progress[vehicle_id].overall_phase = "completed"
            # P2（v7）：成功时同步写 CollectionStatus.status，避免 DB-first 短路掩盖
            await session.execute(
                update(CollectionStatus).where(
                    CollectionStatus.vehicle_id == vehicle_id,
                    CollectionStatus.source == "all",
                ).values(status="completed", error_message=None)
            )
            await session.commit()
    except Exception as e:
        pipeline_phases[vehicle_id] = {"phase": "error", "error": str(e), "_updated_at": datetime.now()}
        # P2（v7）：失败时同步写 CollectionStatus.status="error"，避免 DB-first 短路掩盖为 completed
        try:
            async with _sf() as err_s:
                await err_s.execute(
                    update(CollectionStatus).where(
                        CollectionStatus.vehicle_id == vehicle_id,
                        CollectionStatus.source == "all",
                    ).values(status="error", error_message=str(e)[:200])
                )
                await err_s.commit()
        except Exception:
            pass
    finally:
        # P1-2：finally 复位 CollectionStatus.is_running（与 _run_full_pipeline 同款）
        try:
            async with _sf() as rel:
                await rel.execute(
                    update(CollectionStatus).where(
                        CollectionStatus.vehicle_id == vehicle_id,
                        CollectionStatus.source == "all",
                    ).values(is_running=False)
                )
                await rel.commit()
        except Exception:
            pass
```

    if (data.status === "already_running") {
        this.pipelinePhase = data.phase;
        this.showToast("该车已有任务运行中", "info");
        this.refreshing = false;
        return;
    }
    this._startPolling();
},

// P2-2：error 时不走 _onRefreshCompleted，弹失败
async _pollPipeline() {
    const [data, pd] = await Promise.all([...]);
    if (!data) return;
    this.pipelinePhase = data.phase;
    this.pipelineInfo = data;
    if (pd) {
        this.collectionProgress = pd;
        if (pd.overall_phase === "refreshing" || this.refreshing) {
            this.collectionProgress.platforms = {};  // P2-6：清空平台区
        }
    }
    if (data.phase === "completed" || data.phase === "error") {
        clearInterval(this.pipelinePollTimer);
        this.pipelinePollTimer = null;
        this.loading = false;
        if (this.refreshing) {
            // P2-2：completed 走成功，error 走失败
            if (data.phase === "completed") {
                await this._onRefreshCompleted();
            } else {
                this.showToast("刷新失败：" + (data.error || "未知错误"), "error");
                this.refreshing = false;
            }
        } else {
            // 采集完成（原有逻辑，P2-2 回归验证不破坏）
            if (data.phase === "completed") {
                await this.onVehicleChange();
            } else {
                this.showToast("采集失败", "error");
            }
        }
    }
},

async _onRefreshCompleted() {
    this.showToast("互动数 + 热度已更新", "success");
    await this._reloadAfterRefresh();
    this.refreshing = false;
},

async _reloadAfterRefresh() {
    if (!this.currentVehicleId) return;
    await Promise.all([
        this.loadOverview(),
        this.loadTrend(),
        this.loadTopPosts(),
        this.loadNegativeRadar(),
        this.loadPosts(1),
        this.loadControversialPosts(),
    ]);
    this.renderAllCharts();
    // P1-5：跳过 defect-themes
},

// P0-1：方法首行关菜单
// P1-4：_pollFleet 容错 + 120min 超时兜底
async triggerFleetCollect() {
    this.showOverflow = false;  // P0-1
    if (this.fleetCollecting) return;
    this.fleetCollecting = true;
    this.fleetStartTime = Date.now();  // P1-4：超时兜底
    const data = await this.apiFetch("/api/fleet/collect", { method: "POST" });
    if (!data) { this.fleetCollecting = false; return; }
    
    const msg = `已触发 ${data.started.length} 车采集，跳过 ${data.skipped.length} 车（频率分层）`;
    this.showToast(msg, "success");
    this.openFleet(true);
    
    if (data.started.length > 0) {
        this._startFleetPolling(data.started);
    } else {
        this.fleetCollecting = false;
    }
},

// P1-4：容错——null 视作"未完成继续轮询" + 120min 超时兜底
_startFleetPolling(vids) {
    this.fleetStartedVids = vids;  // P2-1：改名
    if (this.fleetPollTimer) clearInterval(this.fleetPollTimer);
    this.fleetPollTimer = setInterval(() => this._pollFleet(), 5000);
    this._pollFleet();
},

async _pollFleet() {
    // P1-4：120min 超时兜底
    if (Date.now() - this.fleetStartTime > 120 * 60 * 1000) {
        clearInterval(this.fleetPollTimer);
        this.fleetPollTimer = null;
        this.fleetCollecting = false;
        this.showToast("全车采集超时（120min），已强制结束", "error");
        return;
    }
    
    const statuses = await Promise.all(
        this.fleetStartedVids.map(vid => 
            this.apiFetch(`/api/vehicles/${vid}/pipeline-status`, { silent: true })
        )
    );
    // P1-4：null 视作"未完成继续轮询"（不因单轮 null 永久卡死）
    // 只有所有车都返回非 null 且为终态才复位
    const allDone = statuses.every(s => s && (s.phase === "completed" || s.phase === "error"));
    const anyNull = statuses.some(s => !s);
    
    if (allDone && !anyNull) {
        clearInterval(this.fleetPollTimer);
        this.fleetPollTimer = null;
        this.fleetCollecting = false;
        const failed = statuses.filter(s => s.phase === "error").length;
        const msg = `全车采集完成，${statuses.length - failed} 成功 / ${failed} 失败`;
        this.showToast(msg, failed > 0 ? "error" : "success");
        this.openFleet(true);
    }
    // anyNull 或非终态：继续轮询（P1-4 容错）
},
